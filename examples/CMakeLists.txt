cmake_minimum_required(VERSION 3.27)
project(gr4Intro LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(gnuradio4 CONFIG REQUIRED)
find_package(GnuRadioBlockLib CONFIG REQUIRED)
find_package(gnuradio4Blocks CONFIG REQUIRED)
add_executable(reference static-flowgraph/main.cpp)
target_link_libraries(reference PRIVATE gnuradio4::gr-fileio gnuradio4::gr-filter gnuradio4::gnuradio-core gnuradio4::gnuradio-blocklib-core)
add_executable(gain-teaser block-demo/main.cpp)
target_link_libraries(gain-teaser PRIVATE gnuradio4::gr-testing gnuradio4::gnuradio-core gnuradio4::gnuradio-blocklib-core)
enable_testing()
add_test(NAME gain-teaser COMMAND gain-teaser)
set_tests_properties(gain-teaser PROPERTIES TIMEOUT 30)
